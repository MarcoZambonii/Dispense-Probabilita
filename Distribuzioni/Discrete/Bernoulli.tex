\begin{table}[H]
\centering
\renewcommand{\arraystretch}{2}
\begin{tabular}{@{}ll@{}}
\toprule
\textbf{Supporto} 
& $\{0,1\}$ \\

\midrule

\textbf{Funzione di Massa di Probabilità} 
& $\mathbb{P}(X=1)=p,\qquad \mathbb{P}(X=0)=1-p$ \\

\midrule

\textbf{Funzione di Ripartizione}
& $F(k)=
\begin{cases}
0, & k<0,\\[4pt]
1-p, & 0\le k<1,\\[4pt]
1, & k\ge 1.
\end{cases}$ \\[6pt]

\midrule

\textbf{Valore atteso} 
& $\mathbb{E}[X]=p$ \\

\midrule

\textbf{Varianza}
& $Var(X)=p(1-p)$ \\

\midrule

\textbf{Funzione caratteristica}
& $\varphi_X(u)=\mathbb{E}[e^{iuX}]=(1-p)+p\,e^{iu}$ \\

\bottomrule
\end{tabular}
\end{table}
